## 🏗️ Target Architecture

```python
# core/orchestration_v7.py (TARGET: ~100-150 lines)
class OrchestratorV7:
    """
    Lean orchestrator coordinating specialized components.

    Responsibilities (ONLY):
    - Coordinate process_turn flow
    - Delegate to specialized components
    - Maintain FSM state (via StateHandler)
    - Expose public API
    """

    def __init__(self, workspace_path, config, gemini_info, claude_info):
        # Initialize components (composition pattern)
        self.state_handler = StateHandler(...)
        self.guard_pipeline = GuardPipeline(...)
        self.task_router = TaskRouter(...)
        self.task_executor = TaskExecutor(...)
        self.result_handler = ResultHandler(...)
        self.error_handler = ErrorHandler(...)

        # Existing delegated components
        self.context_builder = ContextBuilder(...)
        self.swarm_bridge = SwarmBridge(...)
        # ... etc

    async def process_turn_async(self, user_input=None) -> Dict:
        """Main entry point - orchestrates the flow."""
        # 1. Security: Guard pipeline
        guard_result = self.guard_pipeline.validate_input(user_input)
        if not guard_result.is_safe:
            return self.result_handler.make_error_result(...)

        # 2. Routing: Fast path vs HiveMind
        route = self.task_router.determine_route(user_input, self.state_handler.state)
        if route.is_fast_path:
            return self.result_handler.make_trivial_response(user_input)

        # 3. Execution: Delegate to appropriate handler
        result = await self.task_executor.execute(route, user_input)

        # 4. State transition
        self.state_handler.transition_to(result.next_state)

        return self.result_handler.make_result(result)
```
